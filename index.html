<script>

let selectedPlace;

let map;

const API_URL = "https://script.google.com/macros/s/AKfycbzEOFkrh9MuJChAnfVZXAzBAUmO-WrMQ3wGq6l80gjGRRno79tVEZvdkzTrvackh1NX/exec"; // Ends in /exec


function initAutocomplete() {

  const input = document.getElementById("address");

  const autocomplete = new google.maps.places.Autocomplete(input);

  autocomplete.addListener("place_changed", () => {

    selectedPlace = autocomplete.getPlace();

  });

}


window.onload = initAutocomplete;


function goToMap() {

  if (!selectedPlace || !selectedPlace.geometry) {

    alert("Please select an address from the dropdown list.");

    return;

  }

  document.getElementById("step1").style.display = "none";

  document.getElementById("map").style.display = "block";

  document.getElementById("loader").style.display = "flex";

  initMap();

}


function initMap() {

  const userLoc = selectedPlace.geometry.location;

  map = new google.maps.Map(document.getElementById("map"), {

    center: userLoc,

    zoom: 12,

    disableDefaultUI: true,

    zoomControl: true

  });


  // User location marker

  new google.maps.Marker({

    position: userLoc,

    map: map,

    icon: {

      path: google.maps.SymbolPath.CIRCLE,

      scale: 10,

      fillColor: "#4f46e5",

      fillOpacity: 1,

      strokeWeight: 2,

      strokeColor: "white"

    }

  });


  // Fetch data from Apps Script

  fetch(API_URL)

    .then(res => res.json())

    .then(data => {

      document.getElementById("loader").style.display = "none";

      if(data.error) { console.error(data.error); return; }


      data.forEach(loc => {

        const marker = new google.maps.Marker({

          position: { lat: loc.lat, lng: loc.lng },

          map: map

        });

        marker.addListener("click", () => openPanel(loc));

      });

    })

    .catch(err => {

      console.error("Fetch error:", err);

      document.getElementById("loader").style.display = "none";

    });

}


function openPanel(loc) {

  document.getElementById("panel").style.display = "block";

  document.getElementById("title").innerText = loc.name;

  document.getElementById("addressLink").innerText = "📍 " + loc.address;

  document.getElementById("addressLink").href = "https://www.google.com/maps/search/?api=1&query=" + encodeURIComponent(loc.address);

  document.getElementById("phoneLink").innerText = "📞 " + loc.contact;

  document.getElementById("phoneLink").href = "tel:" + loc.contact.replace(/\D/g,'');

}


function closePanel() {

  document.getElementById("panel").style.display = "none";

}
